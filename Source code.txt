@st.cache_data
def load_data():
    df = pd.read_csv("karnataka_accidents_2021_2025.csv")
    df["date"] = pd.to_datetime(df["date"])
    df["year"] = df["date"].dt.year
    df["severity_num"] = df["severity"].astype(int)
    severity_map = {1: "Minor", 2: "Moderate", 3: "Severe", 4: "Fatal"}
    df["severity_label"] = df["severity_num"].map(severity_map)
    return df

import numpy as np

st.subheader("📈 Future Accident Prediction (Simple Trend Forecast)")

if not filtered.empty:
    if "year" in filtered.columns:
        yearly_trend = filtered.groupby("year").size().reset_index(name="accidents")
        X = yearly_trend["year"].values.reshape(-1, 1)
        y = yearly_trend["accidents"].values

        model = LinearRegression()
        model.fit(X, y)

        next_year = yearly_trend["year"].max() + 1
        y_pred_next = model.predict([[next_year]])[0]
        change_percent = ((y_pred_next - y[-1]) / y[-1]) * 100

        fig, ax = plt.subplots(figsize=(6, 4))
        ax.plot(yearly_trend["year"], y, marker="o", label="Historical Data")
        ax.plot([yearly_trend["year"].max(), next_year], [y[-1], y_pred_next], "--o", color="red", label="Predicted")
        ax.set_xlabel("Year")
        ax.set_ylabel("Accidents")
        ax.set_title("Accident Trend and Prediction")
        ax.legend()
        st.pyplot(fig)

        if change_percent > 0:
            st.success(f"🚗 Predicted {next_year}: {int(y_pred_next)} accidents (+{change_percent:.2f}% increase)")
        else:
            st.info(f"🚗 Predicted {next_year}: {int(y_pred_next)} accidents ({change_percent:.2f}% decrease)")
    else:
        st.warning("Year column not found for prediction.")
else:
    st.info("No data available for prediction.")
